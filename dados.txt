# 7. CASOS DE FEBRE AMARELA HUMANA e EPIZOOTIAS 1994-2026 em Caraguatatuba
# Carregar pacotes
library(readr)
library(dplyr)
library(tidyr)
library(stringr)

# Importar o banco bruto
fa <- read_delim(
  "fa_casoshumanos_1994_2026.csv",
  delim = ";",
  locale = locale(encoding = "Latin1"),
  show_col_types = FALSE
)

# Conferir o banco
View(fa)

# Selecionar os registros cujo município provável
#e infecção (MUN_LPI) é Caraguatatuba

fa_caragua <- fa %>%
  filter(
    str_to_upper(str_trim(MUN_LPI)) == "CARAGUATATUBA"
  )

# Contar o número de casos por ano

fa_caragua_ano <- fa_caragua %>%
  count(ANO_IS, name = "CASOS_HUMANOS")

# Incluir também os anos em que houve ZERO casos

fa_caragua_ano <- fa_caragua_ano %>%
  complete(
    ANO_IS = 1994:2026,
    fill = list(CASOS_HUMANOS = 0)
  )

# Adicionar o nome do município

fa_caragua_ano <- fa_caragua_ano %>%
  mutate(
    MUNICIPIO = "CARAGUATATUBA"
  ) %>%
  select(
    MUNICIPIO,
    ANO = ANO_IS,
    CASOS_HUMANOS
  ) %>%
  arrange(ANO)

# Importar banco bruto
epiz <- read_delim(
  "fa_epizpnh_1994-2026.csv",
  delim = ";",
  locale = locale(encoding = "Latin1"),
  show_col_types = FALSE
)

# Conferir banco
View(epiz)

# Filtrar ocorrências em Caraguatatuba
epiz_caragua <- epiz %>%
  filter(
    str_to_upper(str_trim(MUN_OCOR)) == "CARAGUATATUBA"
  )

# Contar epizootias por ano
epiz_caragua_ano <- epiz_caragua %>%
  count(ANO_OCOR, name = "EPIZOOTIAS")

# Adicionar anos com zero ocorrências
epiz_caragua_ano <- epiz_caragua_ano %>%
  complete(
    ANO_OCOR = 1994:2026,
    fill = list(EPIZOOTIAS = 0)
  )

# Montar tabela final
epiz_caragua_ano <- epiz_caragua_ano %>%
  mutate(
    MUNICIPIO = "CARAGUATATUBA"
  ) %>%
  select(
    MUNICIPIO,
    ANO = ANO_OCOR,
    EPIZOOTIAS
  ) %>%
  arrange(ANO)
